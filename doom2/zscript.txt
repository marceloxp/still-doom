// Still MAP01
// MAP01 reloads itself. Each clear adds 1 to 3 more of the same monsters. Dying does not count.
// The player starts with both shotguns. The super shotgun is in hand from the start.
// Shell boxes restock on the floor every load.
// Outside, just past the shotgun, a power item waits with heavier monsters. The player can skip it.

class StillMAP01Announce : Actor
{
	Default
	{
		+NOINTERACTION
		+NOBLOCKMAP
		+NOGRAVITY
		+NOCLIP
	}

	States
	{
	Spawn:
		TNT1 A 10;
		TNT1 A 0
		{
			A_PrintBold(String.Format("\crCYCLE %d", args[0]));
		}
		Stop;
	}
}

// One stack adds 5 damage to every pellet of either shotgun.
class StillGunPower : Inventory
{
	Default
	{
		Inventory.MaxAmount 1000;
		Inventory.InterHubAmount 1000;
		Inventory.PickupMessage "Shotgun hits harder.";
		Inventory.PickupSound "misc/p_pkup";
		+INVENTORY.FANCYPICKUPSOUND
		Radius 20;
		Height 16;
	}

	States
	{
	Spawn:
		SOUL ABCDCB 6 Bright;
		Loop;
	}

	override bool TryPickup(in out Actor toucher)
	{
		bool ok = Super.TryPickup(toucher);
		if (ok && toucher)
			toucher.A_PrintBold(String.Format("\crSHOTGUN +%d", toucher.CountInv("StillGunPower")));
		return ok;
	}

	override void ModifyDamage(int damage, Name damageType, out int newdamage, bool passive, Actor inflictor, Actor source, int flags)
	{
		newdamage = damage;
		if (passive || !owner || !owner.player || (flags & DMG_NO_ENHANCE))
			return;

		let weap = owner.player.ReadyWeapon;
		if (weap && (weap is "Shotgun" || weap is "SuperShotgun"))
			newdamage = damage + (5 * Amount);
	}
}

class StillMAP01Handler : StaticEventHandler
{
	int cycles;
	int extras;
	bool died;

	override void NewGame()
	{
		cycles = 0;
		extras = 0;
		died = false;
	}

	override void PlayerDied(PlayerEvent e)
	{
		if (level.MapName ~== "MAP01")
			died = true;
	}

	override void WorldUnloaded(WorldEvent e)
	{
		if ((level.MapName ~== "MAP01") && !died)
		{
			cycles++;
			extras += Random(1, 3);
		}
		died = false;
	}

	override void PlayerEntered(PlayerEvent e)
	{
		if (!(level.MapName ~== "MAP01"))
			return;

		let who = players[e.PlayerNumber].mo;
		if (!who)
			return;

		let had = who.FindInventory("SuperShotgun");
		who.GiveInventory("Shotgun", 1);
		who.GiveInventory("SuperShotgun", 1);
		who.GiveInventory("Shell", 20);
		if (!had)
			who.A_SelectWeapon("SuperShotgun");
	}

	override void WorldLoaded(WorldEvent e)
	{
		if (!(level.MapName ~== "MAP01"))
			return;

		int shown = cycles + 1;
		int already = CountMonsters();
		if (extras > 0)
			SpawnExtras(extras);
		int boxes = SpawnAmmo();
		int guards = SpawnPower(extras);
		Announce(shown);
		Console.Printf("Still MAP01: cycle %d, monsters %d", shown, CountMonsters());
		Console.Printf("Still MAP01: originals %d", already);
		Console.Printf("Still MAP01: extras %d", extras);
		Console.Printf("Still MAP01: shell boxes %d", boxes);
		Console.Printf("Still MAP01: guards %d", guards);
	}

	static void Announce(int shown)
	{
		let sign = StillMAP01Announce(Actor.Spawn("StillMAP01Announce", (0, 0, 0), NO_REPLACE));
		if (sign)
			sign.args[0] = shown;
	}

	static int CountMonsters()
	{
		int count = 0;
		ThinkerIterator it = ThinkerIterator.Create("Actor");
		Actor mo = Actor(it.Next());
		while (mo)
		{
			if (mo.bISMONSTER && !mo.bFRIENDLY && mo.health > 0 && !mo.player)
				count++;
			mo = Actor(it.Next());
		}
		return count;
	}

	static int SpawnAmmo()
	{
		int placed = 0;
		PlayerPawn who = players[0].mo;
		if (who)
			placed += Scatter("ShellBox", who, 6, 0);

		Array<Actor> sources;
		ThinkerIterator it = ThinkerIterator.Create("Actor");
		Actor mo = Actor(it.Next());
		while (mo)
		{
			if (mo.bISMONSTER && !mo.bFRIENDLY && mo.health > 0 && !mo.player)
				sources.Push(mo);
			mo = Actor(it.Next());
		}

		for (int i = 0; i < sources.Size(); i++)
			placed += Scatter("ShellBox", sources[i], 2, 3 + i);
		return placed;
	}

	static int Scatter(class<Actor> kind, Actor anchor, int want, int salt)
	{
		int placed = 0;
		for (int n = 0; n < want; n++)
		{
			if (TrySpawnItem(kind, anchor, salt + n))
				placed++;
		}
		return placed;
	}

	// Open floor the anchor can see. A miss is skipped, same rule as the monster copies.
	static bool TrySpawnItem(class<Actor> kind, Actor anchor, int n)
	{
		double floor = anchor.floorz;
		double tall = GetDefaultByType(kind).height;
		vector3 origin = anchor.pos;

		for (int ring = 1; ring <= 6; ring++)
		{
			for (int k = 0; k < 12; k++)
			{
				double ang = anchor.angle + (n * 28) + (k * 30);
				double dist = 64.0 + (ring * 32.0);
				vector2 xy = (origin.x + cos(ang) * dist, origin.y + sin(ang) * dist);
				if (!level.IsPointInLevel((xy.x, xy.y, floor)))
					continue;

				Sector sec = level.PointInSector(xy);
				double z = sec.floorplane.ZatPoint(xy);
				double top = sec.ceilingplane.ZatPoint(xy);
				if (top - z < tall)
					continue;
				if (abs(z - floor) > 24)
					continue;

				let item = Actor.Spawn(kind, (xy.x, xy.y, z), NO_REPLACE);
				if (!item)
					continue;
				if (!item.TestMobjLocation() || !item.CheckSight(anchor, SF_IGNOREVISIBILITY))
				{
					item.Destroy();
					continue;
				}
				return true;
			}
		}
		return false;
	}

	// Grass outside, a few steps south of the shotgun at (320, 368).
	static int SpawnPower(int pressure)
	{
		let item = SpawnLoose("StillGunPower", (400, 200));
		if (!item)
			return 0;

		class<Actor> kind = HeavyKind(pressure);
		int want = Random(1, 3);
		int placed = 0;
		for (int n = 0; n < want; n++)
		{
			if (TrySpawnKind(kind, item, n))
				placed++;
		}
		if (placed == 0 && kind != "Demon")
		{
			if (TrySpawnKind("Demon", item, want))
				placed++;
		}
		return placed;
	}

	// Heavier than the map's zombies and imps. Steps up as the extras pile grows.
	static class<Actor> HeavyKind(int pressure)
	{
		static const class<Actor> kinds[] =
		{
			"Demon",
			"Cacodemon",
			"Revenant",
			"HellKnight",
			"BaronOfHell"
		};
		int step = pressure / 4;
		if (step >= kinds.Size())
			step = kinds.Size() - 1;
		if (step < 0)
			step = 0;
		return kinds[step];
	}

	static Actor SpawnLoose(class<Actor> kind, vector2 xy)
	{
		if (!level.IsPointInLevel((xy.x, xy.y, 0)))
			return null;

		Sector sec = level.PointInSector(xy);
		double z = sec.floorplane.ZatPoint(xy);
		double top = sec.ceilingplane.ZatPoint(xy);
		if (top - z < GetDefaultByType(kind).height)
			return null;

		let mo = Actor.Spawn(kind, (xy.x, xy.y, z), NO_REPLACE);
		if (!mo)
			return null;
		if (!mo.TestMobjLocation())
		{
			mo.Destroy();
			return null;
		}
		return mo;
	}

	static bool TrySpawnKind(class<Actor> kind, Actor anchor, int n)
	{
		double floor = anchor.floorz;
		double tall = GetDefaultByType(kind).height;
		vector3 origin = anchor.pos;

		for (int ring = 1; ring <= 6; ring++)
		{
			for (int k = 0; k < 12; k++)
			{
				double ang = (n * 47) + (k * 30);
				double dist = 72.0 + (ring * 24.0);
				vector2 xy = (origin.x + cos(ang) * dist, origin.y + sin(ang) * dist);
				if (!level.IsPointInLevel((xy.x, xy.y, floor)))
					continue;

				Sector sec = level.PointInSector(xy);
				double z = sec.floorplane.ZatPoint(xy);
				double top = sec.ceilingplane.ZatPoint(xy);
				if (top - z < tall)
					continue;
				if (abs(z - floor) > 24)
					continue;

				let extra = Actor.Spawn(kind, (xy.x, xy.y, z), NO_REPLACE);
				if (!extra)
					continue;
				if (!extra.TestMobjLocation() || !extra.CheckSight(anchor, SF_IGNOREVISIBILITY))
				{
					extra.Destroy();
					continue;
				}
				extra.angle = extra.AngleTo(anchor);
				return true;
			}
		}
		return false;
	}

	static void SpawnExtras(int copies)
	{
		Array<Actor> sources;

		ThinkerIterator it = ThinkerIterator.Create("Actor");
		Actor mo = Actor(it.Next());
		while (mo)
		{
			if (mo.bISMONSTER && !mo.bFRIENDLY && mo.health > 0 && !mo.player)
				sources.Push(mo);
			mo = Actor(it.Next());
		}

		int placed = 0;
		for (int i = 0; i < sources.Size(); i++)
		{
			for (int n = 0; n < copies; n++)
			{
				if (TrySpawn(sources[i], n))
					placed++;
			}
		}
		Console.Printf("Still MAP01: placed %d of %d", placed, sources.Size() * copies);
	}

	// Same species, on open floor the original can see. No spot means skip.
	static bool TrySpawn(Actor source, int n)
	{
		double faceAng = source.angle;
		double floor = source.floorz;
		double tall = source.height;
		vector3 origin = source.pos;

		for (int ring = 1; ring <= 8; ring++)
		{
			for (int k = 0; k < 12; k++)
			{
				double ang = faceAng + (n * 19) + (k * 30);
				double dist = 36.0 + (ring * 28.0);
				vector2 xy = (origin.x + cos(ang) * dist, origin.y + sin(ang) * dist);
				vector3 probe = (xy.x, xy.y, floor);
				if (!level.IsPointInLevel(probe))
					continue;

				Sector sec = level.PointInSector(xy);
				double z = sec.floorplane.ZatPoint(xy);
				double top = sec.ceilingplane.ZatPoint(xy);
				if (top - z < tall)
					continue;
				if (abs(z - floor) > 24)
					continue;

				let extra = Actor.Spawn(source.GetClass(), (xy.x, xy.y, z), NO_REPLACE);
				if (!extra)
					continue;
				if (!extra.TestMobjLocation() || !extra.CheckSight(source, SF_IGNOREVISIBILITY))
				{
					extra.Destroy();
					continue;
				}
				extra.angle = faceAng;
				return true;
			}
		}
		return false;
	}
}
